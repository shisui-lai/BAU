# 上位机系统卡顿崩溃问题分析与解决方案

## 1. 问题现象分析

### 1.1 故障症状
- **正常状态**: 通讯速度显示20-40KB/s，数据正常接收
- **异常状态**: 
  - 通讯速度卡在1500KB/s且数值不变化
  - MQTT连接断开后无法重新连接，处于重连状态
  - UI响应正常但数据接收异常

### 1.2 问题特征
- **触发条件**: 长时间运行（挂机一中午后出现）
- **影响范围**: 数据通讯层面，UI层基本正常
- **恢复难度**: 需要重启应用才能恢复

## 2. 根本原因分析

按影响程度排序识别出以下核心问题：

### 2.1 MQTT子进程管理不完善（影响程度：⭐⭐⭐⭐⭐ 最高）

#### 问题1: 子进程异常处理机制不健全
```javascript
// 当前代码问题 (src/main/mqtt.js:567-578)
try {
  result = parseFun(hex)
} catch (err) {
  console.error(`[PARSE_ERR] ${dataType} len=${len} topic=${topic}`, err)
  return  // 仅跳过，未触发恢复机制
}
```

**问题分析**:
- 数据解析异常时只是跳过消息，未检查异常频率
- 连续解析失败可能导致子进程状态异常
- 缺乏异常累积监控和自动恢复

#### 问题2: MQTT连接状态与实际状态不同步
```javascript
// 当前代码问题 (src/renderer/src/stores/communication/mqttStore.js)
// 连接状态可能显示为"重连中"，但实际子进程已异常
const connectionStatus = ref('disconnected')

// 缺乏深度状态检查机制
function updateConnectionStatus(status) {
  connectionStatus.value = status  // 仅依赖子进程报告，无验证
}
```

**问题分析**:
- 子进程异常时，连接状态可能仍显示为"连接中"或"重连中"
- 缺乏主进程与子进程状态的双向验证机制
- 用户界面状态与实际通讯能力不匹配，误导用户操作

### 2.2 MQTT连接管理健壮性不足（影响程度：⭐⭐⭐⭐ 高）

#### 问题3: 连接状态与数据处理状态不同步
```javascript
// 当前代码问题 (src/main/mqtt.js:532-536)
client.on('message', (topic, payload) => {
  if (!isConnected || !client) {
    return  // 简单跳过，未处理状态不一致
  }
}
```

**问题分析**:
- 连接状态检查过于简单
- 可能存在连接已断开但isConnected仍为true的情况
- 缺乏连接健康度检查机制

#### 问题4: 重连机制缺乏异常检测
```javascript
// 当前代码问题 (src/renderer/src/stores/communication/mqttStore.js:294-321)
const success = await this.connect(true)
if (!success) {
  // 只是增加延迟，未分析失败原因
  this.reconnect.nextDelay = Math.min(
    this.reconnect.nextDelay * RECONNECT_CONFIG.backoffFactor,
    RECONNECT_CONFIG.maxDelay
  )
}
```

**问题分析**:
- 重连失败时未分析具体原因
- 缺乏对子进程状态的检查
- 可能在子进程异常时仍尝试重连

### 2.3 异常处理和恢复机制缺陷（影响程度：⭐⭐⭐ 中等）

#### 问题5: 数据解析异常处理不完善
```javascript
// 当前代码问题 (src/main/mqtt.js:571-576)
try {
  result = parseFun(hex)
} catch (err) {
  console.error(`[PARSE_ERR] ${dataType}`, err)
  // 仅记录错误，无累积监控和恢复机制
  return
}
```

**问题分析**:
- 数据解析错误仅记录日志，无累积统计
- 缺乏解析错误率监控和阈值告警
- 高频解析错误可能导致数据丢失，但系统无感知

#### 问题6: 缺乏系统级健康检查
- 无法检测子进程是否正常工作
- 无法检测数据流是否正常
- 无法自动诊断和恢复异常状态

## 3. 解决方案设计

### 3.1 立即优化措施

#### 方案1: 增强MQTT子进程监控和自动重启

**目标**: 实现子进程健康监控和自动恢复
**实现**: 创建进程管理器，监控子进程状态

```javascript
// 新增文件: src/main/processManager.js
class MQTTProcessManager {
  constructor() {
    this.mqttTask = null
    this.restartCount = 0
    this.maxRestarts = 5
    this.lastRestartTime = 0
    this.minRestartInterval = 30000 // 30秒最小重启间隔
    this.healthCheckInterval = 10000 // 10秒健康检查
    this.lastHeartbeat = 0
    this.healthCheckTimer = null
  }

  startMQTTProcess() {
    try {
      this.mqttTask = fork(forkPath1)
      this.setupProcessMonitoring()
      this.startHealthCheck()
      console.log('[ProcessManager] MQTT子进程已启动，PID:', this.mqttTask.pid)
    } catch (error) {
      console.error('[ProcessManager] 启动MQTT子进程失败:', error)
      this.scheduleRestart()
    }
  }

  setupProcessMonitoring() {
    this.mqttTask.on('error', (error) => {
      console.error('[ProcessManager] MQTT子进程错误:', error)
      this.handleProcessError(error)
    })

    this.mqttTask.on('exit', (code, signal) => {
      console.log('[ProcessManager] MQTT子进程退出，代码:', code, '信号:', signal)
      if (code !== 0) {
        this.handleProcessCrash(code, signal)
      }
    })

    // 监听心跳消息
    this.mqttTask.on('message', (msg) => {
      if (msg.type === 'heartbeat') {
        this.lastHeartbeat = Date.now()
      }
      // 转发其他消息...
    })
  }

  startHealthCheck() {
    this.healthCheckTimer = setInterval(() => {
      const now = Date.now()
      
      // 检查心跳超时
      if (now - this.lastHeartbeat > this.healthCheckInterval * 2) {
        console.warn('[ProcessManager] 子进程心跳超时，准备重启')
        this.restartProcess('heartbeat_timeout')
        return
      }

      // 发送健康检查请求
      if (this.mqttTask && !this.mqttTask.killed) {
        this.mqttTask.send({ cmd: 'HEALTH_CHECK', timestamp: now })
      }
    }, this.healthCheckInterval)
  }

  handleProcessError(error) {
    this.logError('process_error', error)
    this.scheduleRestart()
  }

  handleProcessCrash(code, signal) {
    this.logError('process_crash', { code, signal })
    this.scheduleRestart()
  }

  scheduleRestart() {
    const now = Date.now()
    
    // 检查重启频率限制
    if (now - this.lastRestartTime < this.minRestartInterval) {
      console.warn('[ProcessManager] 重启过于频繁，延迟重启')
      setTimeout(() => this.scheduleRestart(), this.minRestartInterval)
      return
    }

    // 检查最大重启次数
    if (this.restartCount >= this.maxRestarts) {
      console.error('[ProcessManager] 达到最大重启次数，停止自动重启')
      this.notifyRestartFailure()
      return
    }

    this.restartProcess('scheduled')
  }

  restartProcess(reason) {
    console.log(`[ProcessManager] 重启MQTT子进程，原因: ${reason}`)
    
    // 清理旧进程
    if (this.mqttTask && !this.mqttTask.killed) {
      this.mqttTask.kill('SIGTERM')
    }
    
    // 清理定时器
    if (this.healthCheckTimer) {
      clearInterval(this.healthCheckTimer)
    }

    // 重启
    this.restartCount++
    this.lastRestartTime = Date.now()
    
    setTimeout(() => {
      this.startMQTTProcess()
    }, 2000) // 2秒后重启
  }

  logError(type, error) {
    const errorLog = {
      timestamp: new Date().toISOString(),
      type: type,
      error: error.message || error,
      stack: error.stack,
      processInfo: {
        pid: this.mqttTask?.pid,
        restartCount: this.restartCount
      }
    }
    
    console.error('[ProcessManager] 错误日志:', JSON.stringify(errorLog, null, 2))
    // 可以在这里添加错误日志持久化
  }

  notifyRestartFailure() {
    // 通知主窗口重启失败
    if (mainWindow && !mainWindow.isDestroyed()) {
      mainWindow.webContents.send('mqtt-restart-failed', {
        restartCount: this.restartCount,
        maxRestarts: this.maxRestarts
      })
    }
  }

  cleanup() {
    if (this.healthCheckTimer) {
      clearInterval(this.healthCheckTimer)
    }
    if (this.mqttTask && !this.mqttTask.killed) {
      this.mqttTask.kill('SIGTERM')
    }
  }
}

export const processManager = new MQTTProcessManager()
```

#### 方案2: 完善连接健康检查机制

**目标**: 实现连接状态的深度检查和自动恢复

```javascript
// 修改文件: src/main/mqtt.js
// 添加心跳和健康检查机制

let lastHeartbeatSent = 0
let lastMessageReceived = 0
let healthCheckTimer = null
let connectionQuality = 'good' // good, poor, bad

// 启动健康检查
function startHealthCheck() {
  if (healthCheckTimer) return

  healthCheckTimer = setInterval(() => {
    checkConnectionHealth()
  }, 10000) // 每10秒检查一次
}

function checkConnectionHealth() {
  const now = Date.now()

  // 发送心跳到主进程
  process.send({
    type: 'heartbeat',
    data: {
      timestamp: now,
      quality: connectionQuality,
      lastMessage: lastMessageReceived
    }
  })
  lastHeartbeatSent = now

  // 检查消息接收情况
  const timeSinceLastMessage = now - lastMessageReceived

  if (timeSinceLastMessage > 30000) { // 30秒无消息
    connectionQuality = 'bad'
    console.warn('[MQTT Health] 连接质量差，30秒无消息')

    // 尝试重新订阅
    if (client && isConnected) {
      resubscribeTopics()
    }
  } else if (timeSinceLastMessage > 15000) { // 15秒无消息
    connectionQuality = 'poor'
    console.warn('[MQTT Health] 连接质量一般，15秒无消息')
  } else {
    connectionQuality = 'good'
  }

  // 检查客户端状态
  if (client && !client.connected && isConnected) {
    console.error('[MQTT Health] 状态不一致：isConnected=true但client.connected=false')
    isConnected = false
    process.send({ type: 'mqtt-disconnected', data: { reason: 'state_mismatch' } })
  }
}

function resubscribeTopics() {
  if (!currentConfig || !currentConfig.subscribeTopics) return

  console.log('[MQTT Health] 重新订阅主题...')
  currentConfig.subscribeTopics.forEach(topic => {
    client.subscribe(topic, (err) => {
      if (err) {
        console.error(`[MQTT Health] 重新订阅失败: ${topic}`, err)
      } else {
        console.log(`[MQTT Health] 重新订阅成功: ${topic}`)
      }
    })
  })
}

// 修改消息处理函数，记录最后接收时间
function setupMessageHandler() {
  if (!client) return

  client.on('message', (topic, payload) => {
    // 更新最后消息时间
    lastMessageReceived = Date.now()

    // 只有在连接状态下才处理消息
    if (!isConnected || !client) {
      console.warn('[MQTT] 收到消息但连接状态异常，跳过处理')
      return
    }

    // 原有的消息处理逻辑...
    const parts = topic.split('/')
    const suffix = parts.at(-1)
    const blockId = Number(parts[3].slice(1))

    let clusterId = 0
    if (parts.length > 4 && parts[4].startsWith('c')) {
      clusterId = Number(parts[4].slice(1))
    }

    const tRecv = performance.now()
    const dataType = suffix.toUpperCase()
    const buf = payload
    const len = buf.length
    const hex = buf.toString('hex')

    const parseFun = TOPIC_TABLE_MAP[suffix]
    if (!parseFun) {
      return
    }

    let result
    try {
      result = parseFun(hex)
    } catch (err) {
      // 增强错误处理
      handleParseError(dataType, len, topic, hex, err)
      return
    }

    const tParsed = performance.now()
    const { baseConfig, data } = result

    const msg = {
      blockId,
      clusterId,
      dataType: suffix.toUpperCase(),
      topic,
      baseConfig,
      data,
      tRecv,
      tParsed,
      payloadSize: len
    }

    sendToParent(msg)
  })
}

// 增强的解析错误处理
let parseErrorCount = 0
let lastParseErrorReset = Date.now()

function handleParseError(dataType, len, topic, hex, err) {
  parseErrorCount++
  const now = Date.now()

  // 每分钟重置错误计数
  if (now - lastParseErrorReset > 60000) {
    parseErrorCount = 1
    lastParseErrorReset = now
  }

  console.error(
    `[PARSE_ERR] ${dataType} len=${len} topic=${topic}\n` +
    `hex=${hex.slice(0, 40)}...\n` +
    `错误计数: ${parseErrorCount}`,
    err
  )

  // 如果解析错误过多，可能需要重启
  if (parseErrorCount > 50) {
    console.error('[MQTT] 解析错误过多，可能需要重启子进程')
    process.send({
      type: 'mqtt-error',
      data: {
        error: 'Too many parse errors',
        count: parseErrorCount,
        needRestart: true
      }
    })
  }

  // 发送解析错误统计
  process.send({
    type: 'parse-error-stats',
    data: {
      count: parseErrorCount,
      dataType,
      topic,
      timestamp: now
    }
  })
}

// 处理主进程的健康检查请求
process.on('message', (message) => {
  const { cmd, topic, payloadHex, config, type, data, isBackground } = message

  // 响应健康检查
  if (cmd === 'HEALTH_CHECK') {
    process.send({
      type: 'health-response',
      data: {
        timestamp: Date.now(),
        requestTime: message.timestamp,
        isConnected,
        connectionQuality,
        lastMessageReceived,
        parseErrorCount,
        clientState: client ? {
          connected: client.connected,
          reconnecting: client.reconnecting
        } : null
      }
    })
    return
  }

  // 原有的消息处理逻辑...
  if (cmd === 'SET_BACKGROUND_MODE') {
    isBackgroundMode = isBackground
    setBackgroundMode(isBackground)
    return
  }

  // 其他命令处理...
})

// 修改连接函数，添加健康检查
function connectMqtt(config) {
  return new Promise((resolve, reject) => {
    try {
      // 清理旧连接
      if (client) {
        console.log('[MQTT Child] 断开现有连接...')
        client.end(true)
        client.removeAllListeners()
      }

      // 停止旧的健康检查
      if (healthCheckTimer) {
        clearInterval(healthCheckTimer)
        healthCheckTimer = null
      }

      currentConfig = config
      const mqttUrl = `mqtt://${config.host}:${config.port}`

      client = mqtt.connect(mqttUrl, {
        username: config.username,
        password: config.password,
        clientId: config.clientId,
        keepalive: config.keepalive || 30,
        connectTimeout: 5000,
        reconnectPeriod: 0,
        clean: false
      })

      // 连接成功事件
      client.on('connect', () => {
        isConnected = true
        lastMessageReceived = Date.now()
        parseErrorCount = 0
        connectionQuality = 'good'

        console.log(`[MQTT Child] 连接成功，客户端ID: ${config.clientId}`)

        // 订阅主题
        if (config.subscribeTopics && config.subscribeTopics.length > 0) {
          config.subscribeTopics.forEach(topic => {
            client.subscribe(topic, (err) => {
              if (err) {
                console.error(`[MQTT Child] 订阅主题失败: ${topic}`, err)
              } else {
                console.log(`[MQTT Child] 订阅主题成功: ${topic}`)
              }
            })
          })
        }

        setupMessageHandler()
        startHealthCheck() // 启动健康检查

        process.send({ type: 'mqtt-connected', data: { clientId: config.clientId, host: config.host } })
        resolve(true)
      })

      // 其他事件处理保持不变...
      client.on('error', (error) => {
        console.error('[MQTT Child] 连接错误:', error)
        isConnected = false
        connectionQuality = 'bad'
        process.send({ type: 'mqtt-error', data: { error: error.message } })
        reject(error)
      })

      // 连接关闭事件
      client.on('close', () => {
        if (isConnected) {
          isConnected = false
          connectionQuality = 'bad'
          console.log('[MQTT Child] 连接已关闭')
          process.send({ type: 'mqtt-disconnected', data: {} })
        }
      })

      // 离线事件
      client.on('offline', () => {
        console.log('[MQTT Child] 离线')
        connectionQuality = 'poor'
        process.send({ type: 'mqtt-offline', data: {} })
      })

      // 连接超时处理
      setTimeout(() => {
        if (!isConnected) {
          console.error('[MQTT Child] 连接超时 (5秒)')
          if (client) {
            client.end(true)
            client.removeAllListeners()
          }
          reject(new Error('连接超时'))
        }
      }, 5000)

    } catch (error) {
      console.error('[MQTT Child] 连接异常:', error)
      reject(error)
    }
  })
}

// 清理函数
function cleanUp() {
  if (healthCheckTimer) {
    clearInterval(healthCheckTimer)
    healthCheckTimer = null
  }

  flushThrottlers()
  cancelThrottlers()

  if (client) {
    client.end(true)
    client.removeAllListeners()
  }

  process.removeAllListeners('message')
}
```

#### 方案3: 增加系统状态诊断工具

**目标**: 提供实时的系统健康状态监控和诊断

```javascript
// 新增文件: src/renderer/src/composables/utils/useSystemDiagnostics.js
import { ref, computed, onMounted, onUnmounted } from 'vue'
import { useMqttStore } from '@/stores/communication/mqttStore'
import { useDataReceptionStore } from '@/stores/communication/dataReceptionStore'

export function useSystemDiagnostics() {
  const mqttStore = useMqttStore()
  const dataReceptionStore = useDataReceptionStore()

  // 诊断状态
  const diagnostics = ref({
    mqtt: {
      status: 'unknown',
      quality: 'unknown',
      lastHeartbeat: 0,
      parseErrors: 0,
      restartCount: 0
    },
    dataFlow: {
      rate: 0,
      lastData: 0,
      status: 'unknown'
    },
    memory: {
      usage: 0,
      trend: 'stable'
    },
    timer: {
      isActive: false,
      lastExecution: 0,
      intervalAccuracy: 'normal'
    }
  })

  const overallHealth = computed(() => {
    const issues = []

    // MQTT连接检查
    if (diagnostics.value.mqtt.status !== 'connected') {
      issues.push('MQTT连接异常')
    }

    if (diagnostics.value.mqtt.quality === 'bad') {
      issues.push('MQTT连接质量差')
    }

    if (diagnostics.value.mqtt.parseErrors > 10) {
      issues.push('数据解析错误过多')
    }

    // 数据流检查
    if (diagnostics.value.dataFlow.status === 'timeout') {
      issues.push('数据接收超时')
    }

    if (diagnostics.value.dataFlow.rate > 1000) {
      issues.push('数据速率异常')
    }

    // 内存检查
    if (diagnostics.value.memory.usage > 500) {
      issues.push('内存使用过高')
    }

    // 定时器检查
    if (!diagnostics.value.timer.isActive) {
      issues.push('数据速率计算定时器未运行')
    }

    if (diagnostics.value.timer.intervalAccuracy === 'poor') {
      issues.push('定时器执行精度异常')
    }

    return {
      status: issues.length === 0 ? 'healthy' : issues.length < 3 ? 'warning' : 'critical',
      issues
    }
  })

  let diagnosticTimer = null

  // 启动诊断
  function startDiagnostics() {
    diagnosticTimer = setInterval(() => {
      updateDiagnostics()
    }, 5000) // 每5秒更新一次

    // 立即执行一次
    updateDiagnostics()
  }

  // 更新诊断信息
  async function updateDiagnostics() {
    try {
      // 更新MQTT状态
      diagnostics.value.mqtt.status = mqttStore.status

      // 更新数据流状态
      diagnostics.value.dataFlow.rate = dataReceptionStore.dataRate
      diagnostics.value.dataFlow.lastData = dataReceptionStore.lastDataTime
      diagnostics.value.dataFlow.status = dataReceptionStore.status

      // 获取系统信息
      if (window.electronAPI && window.electronAPI.getSystemInfo) {
        const systemInfo = await window.electronAPI.getSystemInfo()
        if (systemInfo.memory) {
          diagnostics.value.memory.usage = systemInfo.memory.heapUsed
        }
        if (systemInfo.throttler) {
          diagnostics.value.throttler = systemInfo.throttler
        }
      }

    } catch (error) {
      console.error('[SystemDiagnostics] 更新诊断信息失败:', error)
    }
  }

  // 手动触发系统恢复
  async function triggerSystemRecovery() {
    console.log('[SystemDiagnostics] 触发系统恢复...')

    try {
      // 1. 重启MQTT连接
      if (mqttStore.status !== 'connected') {
        await mqttStore.disconnect()
        await new Promise(resolve => setTimeout(resolve, 2000))
        await mqttStore.connect()
      }

      // 2. 清理内存
      if (window.electronAPI && window.electronAPI.forceGC) {
        await window.electronAPI.forceGC()
      }

      // 3. 重置数据接收监控
      dataReceptionStore.stopMonitoring()
      await new Promise(resolve => setTimeout(resolve, 1000))
      dataReceptionStore.startMonitoring()

      console.log('[SystemDiagnostics] 系统恢复完成')
      return true

    } catch (error) {
      console.error('[SystemDiagnostics] 系统恢复失败:', error)
      return false
    }
  }

  // 导出诊断报告
  function exportDiagnosticReport() {
    const report = {
```

```javascript
// 新增文件: src/main/recoveryManager.js
class SystemRecoveryManager {
  constructor() {
    this.recoveryStrategies = new Map()
    this.recoveryHistory = []
    this.maxRecoveryAttempts = 3
    this.recoveryWindow = 300000 // 5分钟窗口
    this.isRecovering = false

    this.setupRecoveryStrategies()
  }

  setupRecoveryStrategies() {
    // 策略1: MQTT连接恢复
    this.recoveryStrategies.set('mqtt_connection', {
      name: 'MQTT连接恢复',
      priority: 1,
      execute: async () => {
        console.log('[Recovery] 执行MQTT连接恢复...')

        // 1. 强制断开现有连接
        if (processManager.mqttTask && !processManager.mqttTask.killed) {
          processManager.mqttTask.kill('SIGTERM')
          await new Promise(resolve => setTimeout(resolve, 2000))
        }

        // 2. 重启MQTT子进程
        processManager.restartProcess('recovery_triggered')

        // 3. 等待连接恢复
        return new Promise((resolve) => {
          const checkConnection = setInterval(() => {
            if (processManager.isHealthy()) {
              clearInterval(checkConnection)
              resolve(true)
            }
          }, 1000)

          // 30秒超时
          setTimeout(() => {
            clearInterval(checkConnection)
            resolve(false)
          }, 30000)
        })
      }
    })

    // 策略2: 数据流恢复
    this.recoveryStrategies.set('data_flow', {
      name: '数据流恢复',
      priority: 2,
      execute: async () => {
        console.log('[Recovery] 执行数据流恢复...')

        // 1. 重置数据接收监控
        if (mainWindow && !mainWindow.isDestroyed()) {
          mainWindow.webContents.send('reset-data-reception')
        }

        // 2. 清理限流器
        if (processManager.mqttTask && !processManager.mqttTask.killed) {
          processManager.mqttTask.send({ cmd: 'RESET_THROTTLERS' })
        }

        // 3. 重新订阅主题
        if (processManager.mqttTask && !processManager.mqttTask.killed) {
          processManager.mqttTask.send({ cmd: 'RESUBSCRIBE_TOPICS' })
        }

        return true
      }
    })

    // 策略3: 内存清理恢复
    this.recoveryStrategies.set('memory_cleanup', {
      name: '内存清理恢复',
      priority: 3,
      execute: async () => {
        console.log('[Recovery] 执行内存清理恢复...')

        // 1. 强制垃圾回收
        if (global.gc) {
          global.gc()
        }

        // 2. 清理子进程内存
        if (processManager.mqttTask && !processManager.mqttTask.killed) {
          processManager.mqttTask.send({ cmd: 'FORCE_CLEANUP' })
        }

        // 3. 通知渲染进程清理
        if (mainWindow && !mainWindow.isDestroyed()) {
          mainWindow.webContents.send('force-memory-cleanup')
        }

        return true
      }
    })
  }

  // 检测异常并触发恢复
  async detectAndRecover(anomalyType, severity = 'medium') {
    if (this.isRecovering) {
      console.log('[Recovery] 恢复进程正在进行中，跳过新的恢复请求')
      return false
    }

    // 检查恢复频率限制
    if (!this.canAttemptRecovery(anomalyType)) {
      console.warn('[Recovery] 恢复尝试过于频繁，跳过恢复')
      return false
    }

    this.isRecovering = true

    try {
      console.log(`[Recovery] 检测到异常: ${anomalyType}, 严重程度: ${severity}`)

      // 记录恢复尝试
      this.recordRecoveryAttempt(anomalyType, severity)

      // 根据异常类型选择恢复策略
      const strategies = this.selectRecoveryStrategies(anomalyType, severity)

      // 按优先级执行恢复策略
      for (const strategy of strategies) {
        console.log(`[Recovery] 执行恢复策略: ${strategy.name}`)

        try {
          const success = await strategy.execute()
          if (success) {
            console.log(`[Recovery] 恢复策略 ${strategy.name} 执行成功`)
            this.recordRecoverySuccess(anomalyType, strategy.name)
            return true
          } else {
            console.warn(`[Recovery] 恢复策略 ${strategy.name} 执行失败`)
          }
        } catch (error) {
          console.error(`[Recovery] 恢复策略 ${strategy.name} 执行异常:`, error)
        }
      }

      console.error('[Recovery] 所有恢复策略都失败了')
      this.recordRecoveryFailure(anomalyType)
      return false

    } finally {
      this.isRecovering = false
    }
  }

  selectRecoveryStrategies(anomalyType, severity) {
    const strategies = []

    switch (anomalyType) {
      case 'mqtt_connection_lost':
      case 'mqtt_quality_poor':
        strategies.push(this.recoveryStrategies.get('mqtt_connection'))
        break

      case 'data_flow_timeout':
      case 'data_rate_anomaly':
        strategies.push(this.recoveryStrategies.get('data_flow'))
        if (severity === 'high') {
          strategies.push(this.recoveryStrategies.get('mqtt_connection'))
        }
        break

      case 'memory_high':
      case 'parse_errors_high':
        strategies.push(this.recoveryStrategies.get('memory_cleanup'))
        if (severity === 'high') {
          strategies.push(this.recoveryStrategies.get('data_flow'))
        }
        break

      case 'system_unresponsive':
        // 系统无响应时，执行所有策略
        strategies.push(...Array.from(this.recoveryStrategies.values()))
        break
    }

    // 按优先级排序
    return strategies.sort((a, b) => a.priority - b.priority)
  }

  canAttemptRecovery(anomalyType) {
    const now = Date.now()
    const recentAttempts = this.recoveryHistory.filter(
      record => record.anomalyType === anomalyType &&
               (now - record.timestamp) < this.recoveryWindow
    )

    return recentAttempts.length < this.maxRecoveryAttempts
  }

  recordRecoveryAttempt(anomalyType, severity) {
    this.recoveryHistory.push({
      timestamp: Date.now(),
      anomalyType,
      severity,
      status: 'attempted'
    })

    // 清理旧记录
    this.cleanupRecoveryHistory()
  }

  recordRecoverySuccess(anomalyType, strategyName) {
    const lastRecord = this.recoveryHistory[this.recoveryHistory.length - 1]
    if (lastRecord && lastRecord.anomalyType === anomalyType) {
      lastRecord.status = 'success'
      lastRecord.strategy = strategyName
      lastRecord.completedAt = Date.now()
    }
  }

  recordRecoveryFailure(anomalyType) {
    const lastRecord = this.recoveryHistory[this.recoveryHistory.length - 1]
    if (lastRecord && lastRecord.anomalyType === anomalyType) {
      lastRecord.status = 'failed'
      lastRecord.completedAt = Date.now()
    }
  }

  cleanupRecoveryHistory() {
    const now = Date.now()
    this.recoveryHistory = this.recoveryHistory.filter(
      record => (now - record.timestamp) < this.recoveryWindow * 2
    )
  }

  getRecoveryStats() {
    const now = Date.now()
    const recentRecords = this.recoveryHistory.filter(
      record => (now - record.timestamp) < this.recoveryWindow
    )

    const stats = {
      totalAttempts: recentRecords.length,
      successCount: recentRecords.filter(r => r.status === 'success').length,
      failureCount: recentRecords.filter(r => r.status === 'failed').length,
      isRecovering: this.isRecovering,
      recentAnomalies: recentRecords.map(r => ({
        type: r.anomalyType,
        severity: r.severity,
        status: r.status,
        timestamp: r.timestamp
      }))
    }

    stats.successRate = stats.totalAttempts > 0 ?
      (stats.successCount / stats.totalAttempts * 100).toFixed(1) : 0

    return stats
  }
}

export const recoveryManager = new SystemRecoveryManager()
```

```javascript
import { ref, onMounted, onUnmounted } from 'vue'

export function useMemoryManager() {
  const memoryStats = ref({
    current: 0,
    peak: 0,
    trend: 'stable',
    lastCleanup: 0,
    cleanupCount: 0
  })

  const thresholds = {
    warning: 300, // 300MB
    critical: 500, // 500MB
    emergency: 800 // 800MB
  }

  let monitorTimer = null
  let cleanupTimer = null
  let memoryHistory = []

  // 启动内存监控
  function startMemoryMonitoring() {
    monitorTimer = setInterval(() => {
      updateMemoryStats()
      checkMemoryThresholds()
    }, 10000) // 每10秒检查一次

    // 定期清理（每5分钟）
    cleanupTimer = setInterval(() => {
      performRoutineCleanup()
    }, 300000)
  }

  // 更新内存统计
  async function updateMemoryStats() {
    try {
      if (window.electronAPI && window.electronAPI.getMemoryUsage) {
        const usage = await window.electronAPI.getMemoryUsage()

        memoryStats.value.current = usage.heapUsed
        memoryStats.value.peak = Math.max(memoryStats.value.peak, usage.heapUsed)

        // 记录历史数据
        memoryHistory.push({
          timestamp: Date.now(),
          usage: usage.heapUsed
        })

        // 保持最近1小时的数据
        const oneHourAgo = Date.now() - 3600000
        memoryHistory = memoryHistory.filter(record => record.timestamp > oneHourAgo)

        // 计算趋势
        calculateMemoryTrend()
      }
    } catch (error) {
      console.error('[MemoryManager] 获取内存使用情况失败:', error)
    }
  }

  // 计算内存趋势
  function calculateMemoryTrend() {
    if (memoryHistory.length < 6) {
      memoryStats.value.trend = 'stable'
      return
    }

    const recent = memoryHistory.slice(-6) // 最近6个数据点
    const older = memoryHistory.slice(-12, -6) // 之前6个数据点

    if (older.length === 0) {
      memoryStats.value.trend = 'stable'
      return
    }

    const recentAvg = recent.reduce((sum, r) => sum + r.usage, 0) / recent.length
    const olderAvg = older.reduce((sum, r) => sum + r.usage, 0) / older.length

    const changePercent = ((recentAvg - olderAvg) / olderAvg) * 100

    if (changePercent > 10) {
      memoryStats.value.trend = 'increasing'
    } else if (changePercent < -10) {
      memoryStats.value.trend = 'decreasing'
    } else {
      memoryStats.value.trend = 'stable'
    }
  }

  // 检查内存阈值
  function checkMemoryThresholds() {
    const current = memoryStats.value.current

    if (current > thresholds.emergency) {
      console.error('[MemoryManager] 内存使用达到紧急水平，执行紧急清理')
      performEmergencyCleanup()
    } else if (current > thresholds.critical) {
      console.warn('[MemoryManager] 内存使用达到临界水平，执行深度清理')
      performDeepCleanup()
    } else if (current > thresholds.warning) {
      console.warn('[MemoryManager] 内存使用达到警告水平，执行常规清理')
      performRoutineCleanup()
    }
  }

  // 常规清理
  async function performRoutineCleanup() {
    console.log('[MemoryManager] 执行常规内存清理')

    try {
      // 1. 清理Vue组件缓存
      if (window.__VUE_DEVTOOLS_GLOBAL_HOOK__) {
        // 开发模式下清理Vue DevTools缓存
        window.__VUE_DEVTOOLS_GLOBAL_HOOK__.emit('flush')
      }

      // 2. 清理事件监听器
      cleanupEventListeners()

      // 3. 清理过期数据
      cleanupExpiredData()

      // 4. 请求垃圾回收
      if (window.electronAPI && window.electronAPI.forceGC) {
        await window.electronAPI.forceGC()
      }

      memoryStats.value.lastCleanup = Date.now()
      memoryStats.value.cleanupCount++

      console.log('[MemoryManager] 常规清理完成')

    } catch (error) {
      console.error('[MemoryManager] 常规清理失败:', error)
    }
  }

  // 深度清理
  async function performDeepCleanup() {
    console.log('[MemoryManager] 执行深度内存清理')

    try {
      // 1. 执行常规清理
      await performRoutineCleanup()

      // 2. 清理大型数据结构
      cleanupLargeDataStructures()

      // 3. 重置组件状态
      resetComponentStates()

      // 4. 清理图表和可视化组件
      cleanupChartComponents()

      console.log('[MemoryManager] 深度清理完成')

    } catch (error) {
      console.error('[MemoryManager] 深度清理失败:', error)
    }
  }

  // 紧急清理
  async function performEmergencyCleanup() {
    console.log('[MemoryManager] 执行紧急内存清理')

    try {
      // 1. 执行深度清理
      await performDeepCleanup()

      // 2. 强制清理所有缓存
      if (window.caches) {
        const cacheNames = await window.caches.keys()
        await Promise.all(cacheNames.map(name => window.caches.delete(name)))
      }

      // 3. 清理本地存储
      cleanupLocalStorage()

      // 4. 重启关键服务
      restartCriticalServices()

      console.log('[MemoryManager] 紧急清理完成')

    } catch (error) {
      console.error('[MemoryManager] 紧急清理失败:', error)
    }
  }

  // 清理事件监听器
  function cleanupEventListeners() {
    // 清理MQTT监听器
    if (window.electronAPI) {
      // 移除可能泄漏的监听器
      const events = [
        'mqtt-message', 'mqtt-connected', 'mqtt-disconnected',
        'mqtt-error', 'mqtt-data-heartbeat'
      ]

      events.forEach(event => {
        if (window.electronAPI[`remove${event.replace(/-/g, '')}Listener`]) {
          window.electronAPI[`remove${event.replace(/-/g, '')}Listener`]()
        }
      })
    }
  }

  // 清理过期数据
  function cleanupExpiredData() {
    // 清理故障数据缓存
    if (window.faultDataCache) {
      const now = Date.now()
      const maxAge = 24 * 60 * 60 * 1000 // 24小时

      for (const [key, data] of window.faultDataCache.entries()) {
        if (now - data.timestamp > maxAge) {
          window.faultDataCache.delete(key)
        }
      }
    }

    // 清理参数缓存
    if (window.parameterCache) {
      window.parameterCache.clear()
    }
  }

  // 清理大型数据结构
  function cleanupLargeDataStructures() {
    // 清理单体数据缓存
    if (window.cellDataCache) {
      // 只保留最近的数据
      const keys = Array.from(window.cellDataCache.keys())
      if (keys.length > 100) {
        const toDelete = keys.slice(0, keys.length - 50)
        toDelete.forEach(key => window.cellDataCache.delete(key))
      }
    }
  }

  // 重置组件状态
  function resetComponentStates() {
    // 发送重置事件给各个组件
    window.dispatchEvent(new CustomEvent('memory-cleanup-reset'))
  }

  // 清理图表组件
  function cleanupChartComponents() {
    // 清理ECharts实例
    if (window.echarts) {
      window.echarts.dispose()
    }

    // 清理其他图表库实例
    if (window.chartInstances) {
      window.chartInstances.forEach(chart => {
        if (chart.dispose) chart.dispose()
        if (chart.destroy) chart.destroy()
      })
      window.chartInstances.clear()
    }
  }

  // 清理本地存储
  function cleanupLocalStorage() {
    try {
      // 清理临时数据
      const keysToRemove = []
      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i)
        if (key && (key.startsWith('temp_') || key.startsWith('cache_'))) {
          keysToRemove.push(key)
        }
      }

      keysToRemove.forEach(key => localStorage.removeItem(key))

    } catch (error) {
      console.error('[MemoryManager] 清理本地存储失败:', error)
    }
  }

  // 重启关键服务
  function restartCriticalServices() {
    // 重启数据接收监控
    if (window.dataReceptionStore) {
      window.dataReceptionStore.stopMonitoring()
      setTimeout(() => {
        window.dataReceptionStore.startMonitoring()
      }, 1000)
    }
  }

  // 停止监控
  function stopMemoryMonitoring() {
    if (monitorTimer) {
      clearInterval(monitorTimer)
      monitorTimer = null
    }

    if (cleanupTimer) {
      clearInterval(cleanupTimer)
      cleanupTimer = null
    }
  }

  // 获取内存报告
  function getMemoryReport() {
    return {
      stats: memoryStats.value,
      thresholds,
      history: memoryHistory.slice(-20), // 最近20个数据点
      recommendations: generateRecommendations()
    }

    if (memoryStats.value.trend === 'increasing') {
      recommendations.push('内存使用呈上升趋势，建议增加清理频率')
    }

    if (memoryStats.value.current > thresholds.warning) {
      recommendations.push('当前内存使用较高，建议关闭不必要的页面')
    }

    if (memoryStats.value.cleanupCount > 10) {
      recommendations.push('清理频率较高，可能存在内存泄漏')
    }

    return recommendations
  }

  onMounted(() => {
    startMemoryMonitoring()
  })

  onUnmounted(() => {
    stopMemoryMonitoring()
  })

  return {
    memoryStats,
    thresholds,
    startMemoryMonitoring,
    stopMemoryMonitoring,
    performRoutineCleanup,
    performDeepCleanup,
    performEmergencyCleanup,
    getMemoryReport
  }
```

```javascript
      // 响应健康检查
      process.send({
        type: 'health-response',
        data: {
          timestamp: Date.now(),
          isConnected,
          connectionQuality,
          parseErrorCount
        }
      })
      break

    case 'RESET_THROTTLERS':
      // 重置限流器
      cancelThrottlers()
      break

    case 'RESUBSCRIBE_TOPICS':
      // 重新订阅主题
      resubscribeTopics()
      break

    case 'FORCE_CLEANUP':
      // 强制清理
      if (global.gc) global.gc()
      break
  }
})
```

```javascript
const stabilityTest = {
```
