## 1. Auth Store - 权限状态管理

### 📁 文件位置
`src/renderer/src/stores/auth.js`

### 🎯 功能目的
这是整个权限系统的**核心状态管理中心**，负责：
- 存储当前用户的登录状态
- 提供登录、登出等操作方法
- 供其他组件查询用户权限

### 📝 代码逐行讲解

```javascript
// 第1-3行：导入依赖
import { defineStore } from 'pinia'        // Pinia状态管理库的核心函数
import { ref, computed } from 'vue'       // Vue 3响应式API

// 第5行：定义并导出Store
export const useAuthStore = defineStore('auth', () => {
  // 'auth' 是这个Store的唯一标识符
  // 第二个参数是setup函数，使用Composition API风格
```

#### 状态定义（State）

```javascript
// 第7行：角色状态 - 使用ref创建响应式变量
const role = ref(0)
// ref(0) 创建一个响应式引用，初始值为0
// 访问值需要用 .value，如：role.value
// 三种状态：0=未登录, 1=管理员, 2=访客

// 第11行：用户名状态
const username = ref('')
// 存储当前登录用户的用户名
// 管理员登录时存储实际用户名
// 访客登录时存储固定值'guest'
```

#### 计算属性（Computed）

```javascript
// 第14行：是否已登录
const isLoggedIn = computed(() => role.value !== 0)
// computed创建一个计算属性，会根据依赖自动更新
// 只要role不是0，就表示已登录
// 返回布尔值：true/false

// 第15行：是否是管理员
const isAdmin = computed(() => role.value === 1)
// role等于1时返回true，表示是管理员

// 第16行：是否是访客
const isGuest = computed(() => role.value === 2)
// role等于2时返回true，表示是访客
```

**为什么使用computed？**
- 自动缓存：只有依赖改变时才重新计算
- 响应式：值改变会自动触发视图更新
- 简化代码：其他地方可以直接使用 `authStore.isAdmin`

#### 操作方法（Actions）

```javascript
// 第18-21行：管理员登录方法
function loginAsAdmin(user) {
  role.value = 1              // 设置角色为管理员
  username.value = user       // 保存用户名
}
// 用法：authStore.loginAsAdmin('admin')

// 第23-26行：访客登录方法
function loginAsGuest() {
  role.value = 2              // 设置角色为访客
  username.value = 'guest'    // 固定用户名为'guest'
}
// 用法：authStore.loginAsGuest()

// 第28-31行：登出方法
function logout() {
  role.value = 0              // 重置为未登录状态
  username.value = ''         // 清空用户名
}
// 用法：authStore.logout()
```

#### 返回值（Export）

```javascript
// 第33-42行：返回Store的公共接口
return {
  role,           // 状态：角色值（可读可写）
  username,       // 状态：用户名（可读可写）
  isLoggedIn,     // 计算属性：是否已登录（只读）
  isAdmin,        // 计算属性：是否管理员（只读）
  isGuest,        // 计算属性：是否访客（只读）
  loginAsAdmin,   // 方法：管理员登录
  loginAsGuest,   // 方法：访客登录
  logout          // 方法：登出
}
```

### 🔧 如何使用Auth Store

#### 在Vue组件中使用

```javascript
<script setup>
import { useAuthStore } from '@/stores/auth'

// 1. 获取Store实例
const authStore = useAuthStore()

// 2. 读取状态
console.log(authStore.role)         // 0, 1, 或 2
console.log(authStore.username)     // 用户名字符串
console.log(authStore.isLoggedIn)   // true 或 false
console.log(authStore.isAdmin)      // true 或 false
console.log(authStore.isGuest)      // true 或 false

// 3. 调用方法
authStore.loginAsAdmin('admin')     // 管理员登录
authStore.loginAsGuest()            // 访客登录
authStore.logout()                  // 登出

// 4. 在模板中使用
// <div v-if="authStore.isAdmin">管理员内容</div>
// <div v-else-if="authStore.isGuest">访客内容</div>
</script>
```

#### 在路由守卫中使用

```javascript
import { useAuthStore } from '@/stores/auth'

router.beforeEach((to, from, next) => {
  const authStore = useAuthStore()
  
  if (!authStore.isLoggedIn) {
    // 未登录，跳转到登录页
    return next({ name: 'Login' })
  }
  
  if (authStore.isGuest && to.meta.requiresAdmin) {
    // 访客访问管理员页面，拒绝访问
    return next({ name: 'Home' })
  }
  
  next()
})
```

### 💡 关键知识点

#### 1. Pinia vs Vuex
- **Pinia** 是Vue官方推荐的新一代状态管理库
- 更简单：不需要mutations，直接修改state
- 更好的TypeScript支持
- 支持组合式API（Composition API）

#### 2. ref vs reactive
```javascript
// ref: 用于基本类型和单个对象
const role = ref(0)
role.value = 1  // 需要.value访问

// reactive: 用于复杂对象
const user = reactive({ role: 0, name: '' })
user.role = 1   // 不需要.value
```

我们使用`ref`因为存储的是简单值。

#### 3. computed的优势
```javascript
// 不使用computed（不推荐）
const isAdmin = () => role.value === 1
// 每次调用都重新计算

// 使用computed（推荐）
const isAdmin = computed(() => role.value === 1)
// 自动缓存，只在依赖改变时重新计算
```

## 2. 登录页面 - login.vue

### 📁 文件位置
`src/renderer/src/views/auth/login.vue`

### 🎯 功能目的
提供用户登录界面，支持两种登录方式：
- 管理员登录（需要账号密码）
- 访客登录（直接进入）

### 📝 代码讲解

#### Script部分 - 逻辑层

```javascript
<script setup>
// setup 是Vue 3组合式API的语法
// 代码直接在setup中执行，不需要return

// 1. 导入依赖
import { ref, onMounted } from 'vue'
// ref: 创建响应式变量
// onMounted: 组件挂载后的生命周期钩子

import { useRouter } from 'vue-router'
// 路由实例，用于页面跳转

import { useAuthStore } from '../../stores/auth.js'
// 引入我们刚才讲的Auth Store

// 2. 创建响应式数据
const username = ref('')           // 用户名输入框绑定的值
const password = ref('')           // 密码输入框绑定的值
const errorMessage = ref('')       // 错误提示信息
const rememberMe = ref(true)       // 是否记住密码（默认true）

// 3. 获取实例
const router = useRouter()         // 路由实例
const authStore = useAuthStore()   // Store实例

// 4. 管理员登录逻辑
const onAdminLogin = () => {
  // 清空之前的错误信息
  errorMessage.value = ''
  
  // 验证账号密码（硬编码，实际应该调用API）
  if (username.value === 'admin' && password.value === 'admin') {
    // 验证通过
    
    // 4.1 调用Store的登录方法
    authStore.loginAsAdmin(username.value)
    
    // 4.2 处理"记住密码"功能
    if (rememberMe.value) {
      // 保存到localStorage（浏览器本地存储）
      localStorage.setItem('username', username.value)
      localStorage.setItem('password', password.value)
    } else {
      // 清除localStorage中的数据
      localStorage.removeItem('username')
      localStorage.removeItem('password')
    }
    
    // 4.3 跳转到首页
    router.replace({ name: 'cellData' })
    // replace不会产生历史记录，用户无法后退到登录页
  } else {
    // 验证失败，显示错误信息
    errorMessage.value = '用户名或密码错误'
  }
}

// 5. 访客登录逻辑
const onGuestEnter = () => {
  // 直接调用访客登录方法
  authStore.loginAsGuest()
  // 跳转到首页
  router.replace({ name: 'cellData' })
}

// 6. 组件挂载时执行
onMounted(() => {
  // 尝试从localStorage读取保存的账号密码
  const savedUsername = localStorage.getItem('username')
  const savedPassword = localStorage.getItem('password')
  
  // 如果存在，则自动填充
  if (savedUsername && savedPassword) {
    username.value = savedUsername
    password.value = savedPassword
    rememberMe.value = true
  }
})
</script>
```

### 💡 关键知识点

#### 1. v-model双向绑定原理
```vue
<!-- v-model 是语法糖 -->
<InputText v-model="username" />

<!-- 等价于 -->
<InputText 
  :value="username"
  @input="username = $event.target.value"
/>
```

#### 2. localStorage API
```javascript
// 保存数据
localStorage.setItem('key', 'value')

// 读取数据
const value = localStorage.getItem('key')

// 删除数据
localStorage.removeItem('key')

// 清空所有数据
localStorage.clear()
```

特点：
- 持久化存储，关闭浏览器数据不丢失
- 只能存储字符串，复杂对象需要JSON.stringify
- 同源策略限制

#### 3. router.push vs router.replace
```javascript
// push: 会产生历史记录，可以后退
router.push({ name: 'Home' })

// replace: 不产生历史记录，无法后退
router.replace({ name: 'Home' })
```

登录页使用`replace`，防止用户后退到登录页。

### 📝 代码讲解

#### 导入依赖

```javascript
import { createRouter, createWebHashHistory } from 'vue-router'

import { useAuthStore } from '../stores/auth'
```

#### 路由配置

```javascript
const router = createRouter({
  history: createWebHashHistory(),
  // Hash模式：URL形如 http://localhost:5173/#/login
  // 优点：兼容性好，不需要服务器配置
  // 缺点：URL不够优雅
  
  routes: [
    // 1. 登录页路由
    {
      path: '/login',           // URL路径
      name: 'Login',            // 路由名称（唯一标识）
      component: () => import('@/views/auth/login.vue')
      // 懒加载：只有访问时才加载组件，优化性能
    },
    
    // 2. 主布局路由（嵌套路由）
    {
      path: '/',
      component: () => import('@/layout/AppLayout.vue'),
      // 所有页面共用的布局组件
      
      children: [
        // 2.1 子路由：首页
        {
          path: '',                    // 空路径表示默认子路由
          name: 'cellData',
          component: () => import('@/views/Cluster/cellData.vue'),
          meta: { visibleForGuest: true }
          // meta: 路由元信息，存储自定义数据
          // visibleForGuest: true 表示访客可见
        },
        
        // 2.2 子路由：版本信息（访客可见）
        {
          path: '/Cluster/version',
          name: 'version',
          component: () => import('@/views/Cluster/version.vue'),
          meta: { visibleForGuest: true }
        },
        
        // 2.3 子路由：远程命令（仅管理员）
        {
          path: '/Cluster/Order',
          name: 'Order',
          component: () => import('@/views/Cluster/Order.vue'),
          meta: { visibleForGuest: false }
          // visibleForGuest: false 表示仅管理员可见
        },
        
        // 2.4 子路由：模拟量校准（管理员+密码保护）
        {
          path: '/Cluster/IvCalibration',
          name: 'IvCalibration',
          component: () => import('@/views/Cluster/Calibration.vue'),
          meta: { 
            visibleForGuest: false,      // 仅管理员
            requirePassword: true         // 需要额外密码
          }
        },
        
        // ... 其他路由省略
      ]
    },
    
    // 3. 404重定向
    {
      path: '/:pathMatch(.*)*',    // 匹配所有未定义的路径

    }
  ]
})
```

#### 全局路由守卫

```javascript
router.beforeEach((to, from, next) => {
  // beforeEach: 在每次路由跳转之前执行
  // to: 目标路由对象
  // from: 来源路由对象
  // next: 继续导航的函数
  
  // 1. 获取Auth Store实例
  const authStore = useAuthStore()
  
  // 2. 规则1：访问登录页
  if (to.name === 'Login') {
    // 如果已经登录，直接跳转到首页
    if (authStore.isLoggedIn) {
      return next({ name: 'cellData' })
    }
    // 未登录，正常访问登录页
    return next()
  }

  // 3. 规则2：核心拦截 - 未登录必须跳转登录页
  if (!authStore.isLoggedIn) {
    return next({ name: 'Login' })
  }

  // 4. 规则3：访客访问管理员页面
  if (to.meta.visibleForGuest === false && authStore.isGuest) {
    // 拦截，跳转到首页
    return next({ name: 'cellData' })
  }

  // 5. 其它情况：放行
  return next()
})
```

### 💡 关键知识点

#### 1. 路由模式对比

```javascript
// Hash模式（我们使用的）
import { createWebHashHistory } from 'vue-router'
const router = createRouter({
  history: createWebHashHistory()
})
// URL: http://localhost/#/home
// 优点：不需要服务器配置
// 缺点：URL不美观

  history: createWebHistory()
})
// URL: http://localhost/home
// 优点：URL美观
// 缺点：需要服务器配置重定向
```

#### 2. 路由懒加载

```javascript
// 不推荐：直接导入
import Home from '@/views/Home.vue'
const routes = [{ path: '/', component: Home }]
// 所有组件一次性加载，首屏慢

// 推荐：懒加载
const routes = [{
  path: '/',
  component: () => import('@/views/Home.vue')
}]
// 访问时才加载，首屏快
```

#### 3. 路由守卫的next参数

```javascript
// 1. 放行
next()

// 2. 拒绝（停留在当前页）
next(false)

// 3. 重定向到其他页面
next({ name: 'Login' })
next({ path: '/login' })
next('/login')

// 4. 重定向并传参
next({ name: 'User', params: { id: 123 } })
```

#### 4. 路由元信息（meta）的使用

```javascript
// 定义
const routes = [{
  path: '/admin',
  meta: {
    requiresAuth: true,
    requiresAdmin: true,
    title: '管理员页面'
  }
}]

// 读取

})
```

```javascript
<script setup>
import { ref, computed } from 'vue'

])

const menuItems = computed(() => {
  return rawMenuItems.value
    // 第一步：过滤顶级菜单项
    .filter(item => item.roles.includes(roleKey.value))
    // includes: 检查数组中是否包含某个值
    
    // 第二步：处理分组菜单的子项
    .map(item => {
      if (item.type === 'group' && item.items) {
        // 过滤子项
        return {
          ...item,    // 展开运算符：复制对象的所有属性
          items: item.items.filter(
            subItem => subItem.roles.includes(roleKey.value)
          )
        }
      }
      return item
    })
    
    // 第三步：移除没有子项的空分组
    .filter(item => {
      if (item.type === 'group') {
        return item.items && item.items.length > 0
      }
      return true
    })
})

// 5. 注销函数
function onLogout() {
  authStore.logout()                    // 清除登录状态
  router.replace({ name: 'Login' })     // 跳转到登录页
}
</script>
```

#### 国际化工具（i18n.js）

```javascript
// 导入语言包
import zhCN from '../locales/zh.json'
import enUS from '../locales/en.json'

// 简单的占位实现
export function useI18n() {
  const locale = 'zh-CN'  // 当前语言
  
  // 翻译函数
  function t(key) {
    // key格式: 'login.title'
    const keys = key.split('.')  // ['login', 'title']
    let value = zhCN
    
    // 遍历key路径
    for (const k of keys) {
      if (value && typeof value === 'object') {
        value = value[k]  // 逐层深入
      } else {
        return key  // 找不到，返回key本身
      }
    }
    
    return value || key
  }
  
  return {
    locale,             // 当前语言
    t,                  // 翻译函数
    availableLocales: ['zh-CN', 'en-US']  // 可用语言列表
  }
}

// 导出语言包
export const messages = {
  'zh-CN': zhCN,
  'en-US': enUS
}
```

## 9. 常见问题与解答

### Q1: 为什么刷新页面会丢失登录状态？

**原因**：
- Auth Store 的数据存储在内存中（ref）
- 刷新页面会重新加载JavaScript，内存清空

**解决方案**（可选）：
```javascript
// 在 auth.js 中添加持久化
export const useAuthStore = defineStore('auth', () => {
  // 初始化时从 localStorage 读取
  const storedRole = localStorage.getItem('userRole')
  const role = ref(storedRole ? parseInt(storedRole) : 0)
  
  // 登录时保存到 localStorage
  function loginAsAdmin(user) {
    role.value = 1
    localStorage.setItem('userRole', '1')
  }
  
  // ... 其他代码
})
```

### Q2: 如何添加新的密码保护页面？

**步骤**：
1. 在 `passwords.js` 添加密码配置
2. 在页面组件中添加密码验证逻辑
3. 在路由meta中标记 `requirePassword: true`

### Q3: 如何修改管理员密码？

在 `login.vue` 中修改：
```javascript
if (username.value === 'admin' && password.value === '新密码') {
  // ...
}
```

### Q4: 如何添加新角色（如：运维人员）？

**步骤**：
1. 在 `auth.js` 中添加新角色值（如：role = 3）
2. 添加计算属性 `isOperator`
3. 添加登录方法 `loginAsOperator`
4. 在路由meta和菜单roles中添加 'operator'
5. 在路由守卫中添加相应逻辑

## 10. 最佳实践建议

### 代码组织

1. **单一职责**：每个文件只做一件事
   - auth.js 只管理登录状态
   - passwords.js 只存储密码配置

2. **命名规范**：
   - 组件：PascalCase（如：LoginPage.vue）
   - 函数：camelCase（如：loginAsAdmin）
   - 常量：SCREAMING_SNAKE_CASE（如：PAGE_PASSWORDS）

3. **注释规范**：
   - 复杂逻辑必须注释
   - 公共函数必须写文档注释

### 性能优化

1. **路由懒加载**：使用动态import
2. **计算属性缓存**：使用computed而非methods
3. **v-if vs v-show**：
   - v-if：条件渲染，切换开销大
   - v-show：CSS切换，初始开销大

### 安全建议

1. **密码不要硬编码**：应该从环境变量或配置文件读取
2. **使用HTTPS**：生产环境必须使用
3. **添加会话超时**：长时间无操作自动登出
4. **记录操作日志**：关键操作需要审计
