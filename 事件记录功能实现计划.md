### 步骤1：实现事件记录标志位数据的接收和显示 ⭐ 优先级最高

#### 1.1 在渲染进程中监听事件记录标志位数据

**目标：** 在 `event.vue` 中添加对 `EVENT_RECORD_FLAG_R` 消息的监听

**实现方式：**
```javascript
// 1. 添加请求函数
const requestEventRecordFlag = async () => {
  try {
    const topic = buildTopic('bms/host/s2d/b{block}/event_record_flag_r')
    await window.electronAPI.mqttPublish(topic, 'ff')
  } catch (error) {
    console.error('[EventRecord] 事件记录标志位请求失败:', error)
  }
}

// 2. 添加IPC监听器
let eventFlagListener = null
const registerEventFlagListener = () => {
  if (eventFlagListener) {
    window.electron.ipcRenderer.removeListener('EVENT_RECORD_FLAG_R', eventFlagListener)
  }

  eventFlagListener = (_, mqttMessage) => {
    const { blockId, data } = mqttMessage
    
    // 检查是否是当前选中的堆
    const blockInfo = getCurrentBlockInfo()
    if (!blockInfo || blockInfo.blockNumber !== blockId) {
      return
    }
    
    // 提取事件记录标志位数据
    if (data && Array.isArray(data)) {
      const eventFlagRec = data.find(d => d.class === '事件记录标志位')
      if (eventFlagRec && eventFlagRec.element) {
        eventData.value = eventFlagRec.element || []
      }
    }
  }
  
  window.electron.ipcRenderer.on('EVENT_RECORD_FLAG_R', eventFlagListener)
}

// 3. 添加周期性读取
let eventFlagReadingTimer = null
const startEventFlagPeriodicReading = () => {
  if (eventFlagReadingTimer) {
    clearInterval(eventFlagReadingTimer)
  }
  
  // 立即读取一次
  requestEventRecordFlag()
  
  // 设置定时器，每秒读取一次（或根据实际需求调整）
  eventFlagReadingTimer = setInterval(() => {
    requestEventRecordFlag()
  }, 1000)
}

const stopEventFlagPeriodicReading = () => {
  if (eventFlagReadingTimer) {
    clearInterval(eventFlagReadingTimer)
    eventFlagReadingTimer = null
  }
}
```

### 步骤2：实现事件记录数据的导出功能

#### 2.1 在MQTT子进程中实现事件记录数据的批量读取

**目标：** 在 `mqtt.js` 中添加事件记录数据的批量读取功能

**实现方式：**
```javascript
// 在 mqtt.js 中添加事件记录读取状态管理
let isReadingEvent = false
let eventReadingOffset = 0
let eventReadingTotal = 0
let eventReadingCurrent = 0

// 处理事件记录读取请求
process.on('message', (message) => {
  if (message.cmd === 'START_READ_EVENT') {
    const { offsetRead, totalRead } = message
    startReadingEvent(offsetRead, totalRead)
  } else if (message.cmd === 'CANCEL_READ_EVENT') {
    cancelReadingEvent()
  }
})

// 开始读取事件记录
async function startReadingEvent(offsetRead, totalRead) {
  if (isReadingEvent) {
    return
  }
  
  isReadingEvent = true
  eventReadingOffset = offsetRead
  eventReadingTotal = totalRead
  eventReadingCurrent = 0
  
  // 循环读取事件记录
  for (let i = 0; i < totalRead; i++) {
    if (!isReadingEvent) {
      break
    }
    
    const recordIndex = offsetRead + i
    
    // 1. 发布事件记录偏移量到设备
    const offsetTopic = buildTopic('bms/host/d2s/b{block}/event_record_offset')
    await publishMessage(offsetTopic, recordIndex.toString(16).padStart(4, '0'))
    
    // 2. 等待设备准备数据（轮询）
    // 3. 读取事件记录数据
    const recordTopic = buildTopic('bms/host/s2d/b{block}/event_record_r')
    await publishMessage(recordTopic, 'ff')
    
    // 4. 等待接收数据并解析
    // 5. 发送进度更新
    eventReadingCurrent = i + 1
    process.send({
      type: 'readEventProgress',
      data: {
        current: eventReadingCurrent,
        total: eventReadingTotal
      }
    })
  }
  
  // 发送完成通知
  process.send({
    type: 'readEventCompleted',
    data: {}
  })
  
  isReadingEvent = false
}
```

```javascript
ipcMain.on('start-reading-data-event', (event, { offsetRead, totalRead, blockId }) => {
```

#### 2.3 在渲染进程中实现导出功能

**目标：** 在 `event.vue` 中添加导出功能

**实现方式：**
```javascript
// 导出参数
const valueRead = ref(100)
const flagDownloadAll = ref(false)
const offsetRead = computed(() => {
  if (flagDownloadAll.value) {
    return 0
  }
  const storageCount = eventData.value.find(d => d.label === '事件记录存储数量')?.value || 0
  return Math.max(0, storageCount - valueRead.value)
})

// 开始导出
const confirmExport = () => {
  const blockInfo = getCurrentBlockInfo()
  if (!blockInfo) {
    toast.add({
      severity: 'error',
      summary: '导出失败',
      detail: '请先选择堆',
      life: 3000
    })
    return
  }
  
  const storageCount = eventData.value.find(d => d.label === '事件记录存储数量')?.value || 0
  const totalRead = flagDownloadAll.value ? storageCount : valueRead.value
  
  if (totalRead <= 0) {
    toast.add({
      severity: 'warn',
      summary: '导出失败',
      detail: '导出数量必须大于0',
      life: 3000
    })
    return
  }
  
  window.electron.ipcRenderer.send('start-reading-data-event', {
    offsetRead: offsetRead.value,
    totalRead: totalRead,
    blockId: blockInfo.blockNumber
  })
}

// 取消导出
const cancelExport = () => {
  const blockInfo = getCurrentBlockInfo()
  if (blockInfo) {
    window.electron.ipcRenderer.send('cancelExport', {
      blockId: blockInfo.blockNumber
    })
  }
}

// 监听导出进度
window.electron.ipcRenderer.on('update-readEventProgress', (_, { current, total }) => {
  exportStore.update(current, total)
})

// 监听导出完成
window.electron.ipcRenderer.on('export-completed', () => {
  exportStore.complete()
  toast.add({
    severity: 'success',
    summary: '导出完成',
    detail: '事件记录导出成功',
    life: 3000
  })
})
```

### 步骤3：实现事件记录数据的删除功能

#### 3.1 在MQTT子进程中实现删除功能

**实现方式：**
```javascript
// 处理删除请求
process.on('message', (message) => {
  if (message.cmd === 'DELETE_EVENT_RECORD') {
    const { deleteCount, deleteAll } = message
    deleteEventRecord(deleteCount, deleteAll)
  }
})

// 删除事件记录
async function deleteEventRecord(deleteCount, deleteAll) {
  const deleteTopic = buildTopic('bms/host/d2s/b{block}/event_record_delete')
  
  if (deleteAll) {
    // 全部删除：发送 0xFFFF
    await publishMessage(deleteTopic, 'FFFF')
  } else {
    // 部分删除：发送删除数量
    const deleteValue = deleteCount.toString(16).padStart(4, '0')
    await publishMessage(deleteTopic, deleteValue)
  }
}
```

#### 3.2 在主进程中实现IPC通信

**实现方式：**

```javascript
// 删除参数
const n = ref(1)
const flagDeleteAll = ref(false)
const deleteCount = computed(() => n.value * 16)

// 确认删除
const confirmDelete = () => {
  const blockInfo = getCurrentBlockInfo()
  if (!blockInfo) {
    toast.add({
      severity: 'error',
      summary: '删除失败',
      detail: '请先选择堆',
      life: 3000
    })
    return
  }
  
  confirm.require({
    message: flagDeleteAll.value
      ? '确认删除所有事件记录？'
      : `确认删除 ${deleteCount.value} 条事件记录？`,
    header: '删除确认',
    icon: 'pi pi-exclamation-triangle',
    accept: () => {
      window.electron.ipcRenderer.send('delete-event-record', {
        deleteCount: deleteCount.value,
        deleteAll: flagDeleteAll.value,
        blockId: blockInfo.blockNumber
      })
    }
  })
}
```
